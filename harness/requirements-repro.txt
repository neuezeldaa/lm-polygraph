# TORCH IS DELIBERATELY NOT PINNED TO AN EXACT VERSION.
# An earlier revision pinned torch==2.6.0, which forced a multi-GB downgrade of
# Colab's preinstalled build and broke three things at once:
#   * torchvision (Colab ships 0.26.0+cu128, which expects torch 2.11)
#   * the CUDA build stopped matching the host driver
#   * the kernel kept the pre-install torch while every subprocess got the new
#     one -- a silent in-process/subprocess version split
# Upstream itself only requires torch>=2.6.0, and Colab's build already
# satisfies that and is driver-matched. Reproducibility comes from RECORDING the
# version actually used (harness/provenance.py prints it, and the notebook
# asserts the in-process version matches what is installed on disk), not from
# forcing a downgrade that fights the platform.
torch>=2.6.0

# These are pinned exactly: the transformers version changes generation
# behaviour, and the datasets version changes loader semantics. Both are small
# downloads, so pinning them costs little.
transformers==4.50.0
tokenizers>=0.21,<0.22
datasets==2.21.0            # <3.0: keeps script-based dataset loaders available
accelerate==1.4.0
numpy>=1.26,<2.0            # <2.0 for compatibility with the pinned transformers
hydra-core==1.3.2
omegaconf==2.3.0
safetensors>=0.4.5

# Not used by Qwen2.5 (Qwen2TokenizerFast is BPE: vocab.json + merges.txt, and
# loads no sentencepiece module), but lm-polygraph declares >=0.2.1 and an
# earlier revision of this file pinned 0.2.0 -- below the declared minimum,
# which made the resolver complain about the library not satisfying itself.
sentencepiece>=0.2.1

# Imported at module load by lm_polygraph.generation_metrics
